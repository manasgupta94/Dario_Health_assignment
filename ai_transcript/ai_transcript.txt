```python
def fetch_weather_for_city(city: Dict[str, float]) -> Dict:
    """Fetch weather forecast data for a single city using Open-Meteo."""
    url = (
        "https://api.open-meteo.com/v1/forecast?"
        f"latitude={city['latitude']}&longitude={city['longitude']}"
        "&daily=temperature_2m_max,temperature_2m_min,precipitation_sum,weather_code&"
        "timezone=auto"
    )
    response = requests.get(url, timeout=30)
    response.raise_for_status()
    return response.json()
```

- Sends the GET request using requests.get
- timeout=30 means it waits up to 30 seconds before failing
- `response.raise_for_status()` checks whether the HTTP request was successful
  - if the API fails, it raises an exception
- `response.json()` converts the response into a Python dictionary

```python
def validate_dataset(df: pd.DataFrame) -> Dict:
    """Run standard quality checks and return summary and invalid row list."""
    total_rows = len(df)
    missing_values = int(df.isnull().sum().sum())
    invalid_temperature_rows = int(
        df[(df["temp_max_c"].abs() > 100) | (df["temp_min_c"].abs() > 100)].shape[0]

    duplicate_rows = int(df.duplicated(subset=["city", "date"]).sum())
    invalid_weather_codes = int(df[(df["weather_code"].notna()) & (~df["weather_code"].isin(WEATHER_CODES.keys()))].shape[0])
```

- flags any row where max or min temperature is above 100°C in absolute value

#### invalid_precip_rows
- flags negative rainfall values

#### invalid_weather_codes
- flags codes not present in the weather code dictionary
